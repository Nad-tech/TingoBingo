#pragma once

#include "raylib.h"


//====================================================
// AnimationPriority
//
// Determines which animations are allowed to control
// the Animation object when another animation is
// already playing.
//
// Idle animations are low priority and can be
// interrupted by higher-priority animations such as
// emotions.
//====================================================

enum class AnimationPriority
{
    Idle,
    Emotion
};


//====================================================
// Animation
//
// Controls sprite-sheet animation.
//
// Animation tracks the current frame, timing,
// animation range and playback state.
//
// Individual sprites such as the eyes and mouth use
// this class to control their own animations.
//====================================================

class Animation
{
public:

    // Create an animation with its default state.
    Animation();


    //================================================
    // Initialisation
    //================================================

    // Configure the animation using the dimensions
    // and layout of the sprite sheet.
    void Initialise
    (
        int frameWidth,
        int frameHeight,
        int totalFrames,
        int columns,
        float frameDuration
    );


    //================================================
    // Playback
    //================================================

    // Advance the animation based on elapsed time.
    void Update(float dt);

    // Return the source rectangle for the current
    // frame of the sprite sheet.
    Rectangle GetSourceRectangle() const;


    // Return the dimensions of an individual frame.
    int GetFrameWidth() const;
    int GetFrameHeight() const;


    //================================================
    // Frame Control
    //================================================

    // Set the current animation frame directly.
    void SetFrame(int newFrame);

    // Return the current animation frame.
    int GetFrame() const;

    // Return the total number of frames in the
    // sprite sheet.
    int GetTotalFrames() const;

    // Reset the animation back to its initial state.
    void Reset();

    // Move one frame backwards or forwards.
    void PreviousFrame();
    void NextFrame();


    //================================================
    // Animation Sequences
    //================================================

    // Play a range of frames using the supplied
    // animation priority.
    //
    // Higher-priority animations can interrupt
    // lower-priority animations.
    void Play(
        int startFrame,
        int endFrame,
        AnimationPriority priority
    );

    // Return true while an animation sequence is playing.
    bool IsPlaying() const;

    // Stop the current animation.
    void Stop();


private:

    // Current frame being displayed.
    int frame;


    // Dimensions of an individual sprite frame.
    int frameWidth;
    int frameHeight;


    // Total number of frames in the sprite sheet
    // and number of columns used by the sheet.
    int totalFrames;
    int columns;


    // Tracks elapsed time for the current frame.
    // frameDuration determines how long each frame
    // remains visible.
    float frameTimer;
    float frameDuration;


    // True while an animation sequence is active.
    bool playing;


    // First and last frames of the current animation
    // sequence.
    int startFrame;
    int endFrame;


    // Priority of the animation currently controlling
    // this Animation object.
    AnimationPriority priority;
};

// A world-space debug panel for Raylib 3D scenes.
//
// Usage:
//   DebugPanel::Instance().Initialise();
//   DebugPanel::Instance().SetPosition({0.0f, 4.0f, 0.0f});
//   DebugPanel::Instance().Print("Blinking", true);
//   DebugPanel::Instance().Print("Eye scale", 0.42f);
//   ...
//   DebugPanel::Instance().Draw3D(camera); // inside BeginMode3D()/EndMode3D()
//   DebugPanel::Instance().Shutdown();     // before closing the window
//
// The panel uses a RenderTexture2D to draw normal 2D text, then displays
// that texture on a camera-facing 3D quad. Print() updates an existing label
// rather than adding duplicate rows.

class DebugPanel
{
public:
    // Returns the one shared DebugPanel instance. Any class can use this
    // without receiving a DebugPanel reference through its constructor.
    static DebugPanel& Instance();

    ~DebugPanel();

    // This is a singleton: prevent accidental copies or extra instances.
    DebugPanel(const DebugPanel&) = delete;
    DebugPanel& operator=(const DebugPanel&) = delete;
    DebugPanel(DebugPanel&&) = delete;
    DebugPanel& operator=(DebugPanel&&) = delete;

    // Call after InitWindow(), before using Print() or Draw3D().
    void Initialise();
    void Shutdown();

    void SetPosition(Vector3 position);
    Vector3 GetPosition() const;

    void SetWorldSize(float width, float height);
    void SetTextStyle(int fontSize, int lineSpacing, Color textColour);

    void SetVisible(bool visible);
    bool IsVisible() const;

    // Set or update a row. Supported types: string, const char*, float,
    // double, int, unsigned int, bool, Vector2, and Vector3.
    void Print(const std::string& label, const std::string& value);
    void Print(const std::string& label, const char* value);
    void Print(const std::string& label, float value);
    void Print(const std::string& label, double value);
    void Print(const std::string& label, int value);
    void Print(const std::string& label, unsigned int value);
    void Print(const std::string& label, bool value);
    void Print(const std::string& label, Vector2 value);
    void Print(const std::string& label, Vector3 value);

    void Clear();

    // Call inside BeginMode3D(camera) / EndMode3D().
    void Draw3D(const Camera3D& camera);

private:
    DebugPanel() = default;

    struct Entry
    {
        std::string label;
        std::string value;
    };

    std::vector<Entry> entries;

    RenderTexture2D renderTexture{};
    bool initialised = false;
    bool visible = true;

    Vector3 position = { 0.0f, 3.0f, 0.0f };
    float worldWidth = 2.8f;
    float worldHeight = 1.6f;

    int textureWidth = 1024;
    int textureHeight = 512;
    int fontSize = 60;
    int lineSpacing = 36;
    int padding = 18;

    Color textColour = LIME;
    Color panelColour = { 12, 18, 24, 220 };
    Color borderColour = { 90, 180, 130, 255 };

    void SetValue(const std::string& label, const std::string& value);
    void RefreshTexture();
};

#pragma once

#include "raylib.h"
#include "raymath.h"

struct MyTransform3D
{
    Vector3 position = { 0.0f, 0.0f, 0.0f };

    // Axis around which the model rotates.
    Vector3 rotationAxis = { 0.0f, 0.0f, 0.0f };

    // Rotation angle in degrees.
    float rotationAngle = 0.0f;

    // Scale along the X, Y and Z axes.
    Vector3 scale = { 1.0f, 1.0f, 1.0f };
};

inline MyTransform3D MakeChildTransform(
    const MyTransform3D& parent,
    const MyTransform3D& child)
{
    MyTransform3D result{};

    // 1. Scale the child's local position.
    Vector3 scaledOffset = Vector3Multiply(
        child.position,
        parent.scale
    );

    // 2. Convert the parent's rotation to a quaternion.
    Quaternion parentRotation = QuaternionFromAxisAngle(
        parent.rotationAxis,
        parent.rotationAngle * DEG2RAD
    );

    // 3. Rotate the child's offset around the parent.
    Vector3 rotatedOffset = Vector3RotateByQuaternion(
        scaledOffset,
        parentRotation
    );

    // 4. Calculate the child's world position.
    result.position = Vector3Add(
        parent.position,
        rotatedOffset
    );

    // 5. Combine rotations.
    Quaternion childRotation = QuaternionFromAxisAngle(
        child.rotationAxis,
        child.rotationAngle * DEG2RAD
    );

    Quaternion combinedRotation = QuaternionMultiply(
        parentRotation,
        childRotation
    );

    // 6. Convert the combined rotation back to axis-angle.
    Vector3 axis;
    float angle;

    QuaternionToAxisAngle(
        combinedRotation,
        &axis,
        &angle
    );

    result.rotationAxis = axis;
    result.rotationAngle = angle * RAD2DEG;

    // 7. Combine scales.
    result.scale = Vector3Multiply(
        parent.scale,
        child.scale
    );

    return result;
}

//====================================================
// SpeechController
//
// Manages Tingo's speech generation and audio playback.
//
// Speech generation is performed on a worker thread
// using Piper/FFmpeg, while Raylib audio operations
// are handled on the main thread.
//
// Speech lifecycle:
//
//     Idle → Generating → Playing → Idle
//
// Generating = Piper/FFmpeg are creating the WAV file.
// Playing    = Raylib is playing the generated WAV.
// Idle       = No speech is currently active.
//====================================================

class SpeechController
{
public:

    //================================================
    // State
    //
    // Represents the current stage of the speech
    // lifecycle.
    //================================================

    enum class State
    {
        Idle,
        Generating,
        Playing
    };


    //================================================
    // Update
    //
    // Runs on the main thread.
    //
    // Checks whether the worker thread has generated
    // the WAV file and, if so, loads and plays it
    // through Raylib.
    //
    // Also detects when Raylib has finished playing
    // the speech.
    //================================================

    void Update();


    //================================================
    // Speak
    //
    // Starts a new speech request.
    //
    // Speech generation is performed on a worker
    // thread so Piper/FFmpeg do not block the
    // main game loop.
    //================================================

    void Speak(const std::string& text);


    //================================================
    // Destructor
    //
    // Requests the speech worker to stop and joins
    // the worker thread before the controller is
    // destroyed.
    //================================================

    ~SpeechController();


    //================================================
    // SoundLoaded
    //
    // Returns true when the generated speech WAV
    // is currently loaded into a Raylib Sound.
    //================================================

    bool SoundLoaded();


    //================================================
    // SpeechFinished
    //
    // Returns true when the complete speech lifecycle
    // has finished.
    //
    // This remains false while speech is generating
    // or playing, and becomes true once playback has
    // finished and the controller returns to Idle.
    //================================================

    bool SpeechFinished();


private:

    // Current stage of the speech lifecycle.
    State state = State::Idle;


    //================================================
    // Worker Thread Communication
    //================================================

    // Set by the main thread when shutdown has been
    // requested.
    //
    // The speech worker checks this flag while
    // generating speech and stops when requested.
    std::atomic<bool> stopRequested{false};


    // Set by the worker thread when Speech::Speak()
    // has successfully generated the WAV file.
    //
    // The main thread checks this flag in Update().
    std::atomic<bool> wavReady{false};


    // Worker thread responsible for speech generation.
    std::thread speechThread;


    //================================================
    // Raylib Audio
    //================================================

    // The generated speech audio loaded by Raylib.
    Sound speechSound{};


    // True while speechSound contains a loaded
    // Raylib Sound.
    bool soundLoaded = false;


    // Tracks whether the complete speech request has
    // finished.
    //
    // false = generating or playing
    // true  = speech has completely finished
    //
    // This is separate from SoundIsPlaying() because
    // speech can be not playing while it is still
    // being generated.
    bool speechFinished = true;
};

//====================================================
// Animation.cpp
//
// Manages sprite sheet animations.
//
// The Animation class tracks the current frame,
// advances animations over time and calculates the
// correct source rectangle for rendering.
//====================================================

#include "Animation.h"

// Initialise the animation with default values.
Animation::Animation()
{
    frame = 0;

    frameWidth = 0;
    frameHeight = 0;

    totalFrames = 1;
    columns = 1;

    frameTimer = 0.0f;
    frameDuration = 0.1f;

    playing = false;

    startFrame = 0;
    endFrame = 0;
}

// Configure the animation using information from
// the sprite sheet.
void Animation::Initialise(
    int width,
    int height,
    int frames,
    int cols,
    float duration)
{
    frameWidth = width;
    frameHeight = height;

    totalFrames = frames;
    columns = cols;

    frameDuration = duration;

    // Reset the animation.
    frame = 0;
    frameTimer = 0.0f;
}

// Advance the animation based on elapsed time.
void Animation::Update(float dt)
{
    // Do nothing if the animation isn't currently playing.
    if (!playing)
        return;

    // Count the elapsed time since the last frame.
    frameTimer += dt;

    // Wait until enough time has passed.
    if (frameTimer < frameDuration)
        return;

    // Remove one frame's worth of elapsed time.
    // Keeping any leftover time prevents the animation
    // from slowing down if frame times vary.
    frameTimer -= frameDuration;

    // Advance to the next frame.
    frame++;

    // Stop once the animation reaches the end frame.
    if (frame > endFrame)
    {
        frame = startFrame;
        playing = false;
    }
}

// Return the source rectangle for the current frame
// within the sprite sheet.
Rectangle Animation::GetSourceRectangle() const
{
    int column = frame % columns;
    int row = frame / columns;

    return
    {
        (float)(column * frameWidth),
        (float)(row * frameHeight),
        (float)frameWidth,
        (float)frameHeight
    };
}

// Reset the animation to the first frame.
void Animation::Reset()
{
    frame = 0;
    frameTimer = 0.0f;
}

// Set the current animation frame.
void Animation::SetFrame(int newFrame)
{
    frame = newFrame % totalFrames;
    frameTimer = 0.0f;
}

// Move to the previous frame.
void Animation::PreviousFrame()
{
    frame--;

    // Wrap around to the last frame.
    if (frame < 0)
        frame = totalFrames - 1;
}

// Move to the next frame.
void Animation::NextFrame()
{
    // Wrap around to the first frame.
    frame = (frame + 1) % totalFrames;
}

int Animation::GetFrameWidth() const
{
    return frameWidth;
}

int Animation::GetFrameHeight() const
{
    return frameHeight;
}

int Animation::GetFrame() const
{
    return frame;
}

int Animation::GetTotalFrames() const
{
    return totalFrames;
}

// Play a section of the sprite sheet once.
void Animation::Play(int start, int end, AnimationPriority newPriority)
{
    // Don't interrupt an animation with an equal
    // or higher priority.
    if (playing && static_cast<int>(newPriority) <= static_cast<int>(priority))
        return;

    startFrame = start;
    endFrame = end;

    // Start playback from the first frame.
    frame = startFrame;
    frameTimer = 0.0f;

    priority = newPriority;
    playing = true;
}

// Returns true while an animation is playing.
bool Animation::IsPlaying() const
{
    return playing;
}

void Animation::Stop()
{
    playing = false;
}

DebugPanel::~DebugPanel()
{
    // Raylib resources should be released while the Raylib window/context
    // still exists. Prefer calling Shutdown() explicitly before CloseWindow().
    // The destructor is a convenience for normal object lifetime ordering.
    Shutdown();
}

void DebugPanel::Initialise()
{
    if (initialised || !IsWindowReady())
        return;

    renderTexture = LoadRenderTexture(textureWidth, textureHeight);
    initialised = renderTexture.id != 0 && renderTexture.texture.id != 0;

    if (initialised)
    {
        SetTextureFilter(renderTexture.texture, TEXTURE_FILTER_BILINEAR);
        RefreshTexture();
    }
    else
    {
        if (renderTexture.id != 0)
            UnloadRenderTexture(renderTexture);

        renderTexture = {};
        TraceLog(LOG_WARNING, "DebugPanel: render texture allocation failed");
    }
}

void DebugPanel::Shutdown()
{
    if (initialised)
    {
        UnloadRenderTexture(renderTexture);
        renderTexture = {};
        initialised = false;
    }
}

void DebugPanel::SetValue(const std::string& label, const std::string& value)
{
    for (Entry& entry : entries)
    {
        if (entry.label == label)
        {
            if (entry.value != value)
            {
                entry.value = value;
                RefreshTexture();
            }
            return;
        }
    }

    entries.push_back({ label, value });
    RefreshTexture();
}

void DebugPanel::Clear()
{
    entries.clear();
    RefreshTexture();
}

void DebugPanel::RefreshTexture()
{
    if (!initialised)
        return;

    BeginTextureMode(renderTexture);
    ClearBackground(panelColour);

    // Simple border around the panel.
    DrawRectangleLinesEx(
        { 1.0f, 1.0f, static_cast<float>(textureWidth - 2),
          static_cast<float>(textureHeight - 2) },
        3.0f,
        borderColour
    );

    int y = padding;
    const int availableBottom = textureHeight - padding;

    for (const Entry& entry : entries)
    {
        if (y + fontSize > availableBottom)
            break;

        const std::string output = entry.label + ": " + entry.value;

        DrawText(
            output.c_str(),
            padding,
            y,
            fontSize,
            textColour
        );

        y += lineSpacing;
    }

    EndTextureMode();
}

void DebugPanel::Draw3D(const Camera3D& camera)
{
    if (!visible || !initialised)
        return;

    // raylib's billboard path builds the camera-facing quad and manages its
    // texture and render state. A negative source height corrects the
    // vertical inversion of RenderTexture2D textures.
    DrawBillboardPro(
        camera,
        renderTexture.texture,
        { 0.0f, 0.0f, static_cast<float>(textureWidth), -static_cast<float>(textureHeight) },
        position,
        { 0.0f, 1.0f, 0.0f },
        { worldWidth, worldHeight },
        { worldWidth * 0.5f, worldHeight * 0.5f },
        0.0f,
        WHITE
    );
}

    InitWindow(10, 10, "TingoBingo 3D");

    InitAudioDevice();

    int width = GetMonitorWidth(monitor);
    int height = GetMonitorHeight(monitor);

void Game::Run()
{
    while (!WindowShouldClose())
    {
        float dt = GetFrameTime();

        world.Update(dt);

// Draw a screen-space widget showing the camera's orientation.
void GameWorld::DrawOrientationGizmo()
{
    float x = screenDimensions.screenWidth;

    const Vector2 centre = {
    static_cast<float>(x - 100),
    100.0f
};
    const float axisLength = 35.0f;

    // Calculate the direction from the camera towards its target.
    Vector3 forward = Vector3Normalize(
        Vector3Subtract(camera.target, camera.position)
    );

    // Calculate the camera's right and up directions.
    Vector3 right = Vector3Normalize(
        Vector3CrossProduct(forward, camera.up)
    );

    Vector3 up = Vector3Normalize(
        Vector3CrossProduct(right, forward)
    );

    // World axes and their colours.
    const Vector3 axes[] = {
        { 1.0f, 0.0f, 0.0f },  // X
        { 0.0f, 1.0f, 0.0f },  // Y
        { 0.0f, 0.0f, 1.0f }   // Z
    };

    const Color colours[] = {
        RED,
        GREEN,
        BLUE
    };

    const char* labels[] = { "X", "Y", "Z" };

    // Draw each world axis from the widget's centre.
    for (int i = 0; i < 3; i++)
    {
        const Vector3& axis = axes[i];

        // Project the world axis onto the camera's screen directions.
        float screenX = Vector3DotProduct(axis, right);
        float screenY = -Vector3DotProduct(axis, up);

        Vector2 end = {
            centre.x + screenX * axisLength,
            centre.y + screenY * axisLength
        };

        DrawLineEx(centre, end, 3.0f, colours[i]);

        // Place the axis label just beyond the line's endpoint.
        Vector2 labelPosition = {
            end.x + screenX * 8.0f - 5.0f,
            end.y + screenY * 8.0f - 6.0f
        };

        DrawText(
            labels[i],
            static_cast<int>(labelPosition.x),
            static_cast<int>(labelPosition.y),
            14,
            colours[i]
        );
    }

    // Mark the widget's centre.
    DrawCircleV(centre, 3.0f, DARKGRAY);
}

void GameWorld::HandleInput(float dt)
{
    const float moveSpeed = 3.0f;

    Vector3 direction = { 0.0f, 0.0f, 0.0f };

    if (IsKeyDown(KEY_O))    direction.z -= 1.0f;
    if (IsKeyDown(KEY_L))  direction.z += 1.0f;
    if (IsKeyDown(KEY_K))  direction.x -= 1.0f;
    if (IsKeyDown(KEY_SEMICOLON)) direction.x += 1.0f;
    if (IsKeyDown(KEY_R)) direction.y += 1.0f;
    if (IsKeyDown(KEY_F)) direction.y -= 1.0f;

    if (Vector3Length(direction) > 0.0f)
    {
        direction = Vector3Normalize(direction);

        tingo.GetTransform().position.x += direction.x * moveSpeed * dt;
        tingo.GetTransform().position.z += direction.z * moveSpeed * dt;
        tingo.GetTransform().position.y += direction.y * moveSpeed * dt;
    }

    tingo.SetTransform(tingo.GetTransform());
}

#include "SpeechController.h"
#include "SpeechGenerator.h"
#include <iostream>

//====================================================
// Update
//
// Runs on the main thread.
//
// Handles communication between the speech worker
// thread and the Raylib audio system.
//
// The worker thread is responsible for generating
// the WAV file.
//
// The main thread is responsible for loading and
// playing the WAV through Raylib.
//
// Speech lifecycle:
//
//     Generating → Playing → Idle
//
// speechFinished remains false while speech is
// generating or playing, and becomes true when
// playback has completely finished.
//====================================================

void SpeechController::Update()
{
    //================================================
    // WAV READY
    //
    // The worker thread sets wavReady to true when
    // Speech::Speak() has successfully generated
    // the WAV file.
    //
    // Because wavReady is atomic, it can safely be
    // written by the worker thread and read by the
    // main thread.
    //================================================

    if(wavReady)
    {
        std::cout << "WAV ready - starting playback\n";

        // Reset the flag so the same WAV is not
        // loaded and played again on the next frame.
        wavReady = false;


        // Load the generated WAV into a Raylib Sound.
        //
        // Raylib audio operations are performed on
        // the main thread rather than the worker thread.
        speechSound =
            LoadSound("./assets/audio/voice/speech.wav");

        soundLoaded = true;


        // Start playing the generated speech.
        PlaySound(speechSound);
        std::cout << "PlaySound called\n";


        // Speech generation is complete and the
        // generated audio has now started playing.
        state = State::Playing;


        // The complete speech request is not finished
        // yet because the sound is currently playing.
        speechFinished = false;
    }


    //================================================
    // PLAYBACK FINISHED
    //
    // Check whether speech is currently playing
    // and Raylib has finished the sound.
    //
    // IsSoundPlaying() returns false once playback
    // has completed.
    //
    // IsSoundPlaying() is a Raylib function that checks
    // whether the Sound is currently being played.
    //================================================

    if(state == State::Playing &&
       !IsSoundPlaying(speechSound))
    {
        // The audio has finished playing, so the
        // Raylib Sound resource can now be released.
        UnloadSound(speechSound);


        // Keep track of the fact that there is no
        // longer a loaded speech sound.
        soundLoaded = false;


        // The complete speech lifecycle has finished.
        //
        // Returning to Idle allows the controller to
        // accept another speech request.
        state = State::Idle;


        // Tell the RobotBrain that the entire speech
        // request has now finished.
        speechFinished = true;
    }
}


//====================================================
// Speak
//
// Starts a new speech request.
//
// Speech generation is performed on a worker thread
// so Piper and FFmpeg do not block the main game loop.
//
// The controller moves into the Generating state
// immediately while the worker creates the WAV file.
//====================================================

void SpeechController::Speak(const std::string& text)
{
    std::cout << "Speak called: " << text << '\n';

    //================================================
    // CHECK WHETHER SPEECH CAN START
    //
    // A new speech request can only begin when the
    // controller is Idle.
    //
    // This prevents another worker thread from being
    // created while speech is already generating or
    // playing.
    //================================================

    if (state != State::Idle)
    {
        return;
    }


    //================================================
    // JOIN PREVIOUS WORKER
    //
    // A std::thread remains joinable even after the
    // function running on it has finished.
    //
    // Join the previous worker before reusing the
    // speechThread object.
    //================================================

    if(speechThread.joinable())
    {
        // Wait for the previous worker to finish and
        // safely join it with the main thread.
        speechThread.join();
    }


    //================================================
    // START GENERATION
    //================================================

    // Tell the controller that speech generation is
    // now in progress.
    //
    // This prevents another Speak() request from
    // starting while the current request is active.
    state = State::Generating;


    // The speech request has started, so it cannot be
    // considered finished while the WAV is generating.
    //
    // SpeechFinished must be set to false when a new speech
    // request begins.
    //
    // Otherwise RobotBrain could see the initial true value
    // while speech is still being generated and incorrectly
    // assume that speech has already finished.
    speechFinished = false;


    //================================================
    // CREATE WORKER THREAD
    //================================================

    // Create and start the worker thread.
    //
    // The lambda contains the code executed by the
    // worker thread.
    //
    // [this, text] means:
    //
    // this
    //     Gives the lambda access to this
    //     SpeechController object.
    //
    // text
    //     Makes a copy of the speech text so the
    //     worker has its own copy to use.
    speechThread = std::thread([this, text]()
    {
        //================================================
        // GENERATE SPEECH
        //
        // Speech::Speak() runs Piper and FFmpeg on
        // the worker thread.
        //
        // stopRequested is shared between the main
        // thread and worker thread and allows the
        // generation process to be cancelled.
        //================================================

        bool success =
            SpeechGenerator::GenerateSpeech(text, stopRequested);

        //================================================
        // GENERATION RESULT
        //================================================

        if (success)
        {
            // The WAV has successfully been generated.
            //
            // Tell the main thread that it can now
            // load and play the WAV.
            wavReady = true;
        }
        else
        {
            // Speech generation failed or was cancelled.
            //
            // No audio will be played, so return the
            // controller to the Idle state.
            state = State::Idle;
        }
    });
}


//====================================================
// Destructor
//
// Called when SpeechController is destroyed.
//
// Ensures the speech worker is safely stopped and
// joined before the SpeechController object disappears.
//====================================================

SpeechController::~SpeechController()
{
    //================================================
    // REQUEST WORKER SHUTDOWN
    //================================================

    // Tell Speech::Speak() that shutdown has been
    // requested.
    //
    // Speech::Speak() periodically checks this flag
    // while Piper and FFmpeg are running.
    stopRequested = true;


    //================================================
    // JOIN WORKER
    //================================================

    // Check whether the worker thread still needs
    // to be joined.
    if (speechThread.joinable())
    {
        // Wait for the worker to finish before
        // destroying the SpeechController.
        //
        // This prevents the worker from accessing
        // the controller after it has been destroyed.
        speechThread.join();
    }
}

//====================================================
// Speech.cpp
//
// Generates speech audio using Piper and FFmpeg.
//
// Piper receives text through a Windows stdin pipe.
// This avoids shell execution and means special
// characters in the spoken text are not interpreted
// by the command shell.
//
// Piper creates speech_raw.wav.
//
// FFmpeg then applies the selected audio effects
// and creates the final speech.wav.
//
// All project paths are relative to the application's
// current working directory, so the project does not
// depend on a specific developer's machine path.
//====================================================

#include "SpeechGenerator.h"

#include <filesystem>
#include <iostream>

#include "windows.h"


//====================================================
// Helper: CloseProcessHandles
//
// Safely closes the handles returned by
// CreateProcessA().
//
// The process itself has already finished or has
// been terminated before this function is called.
//====================================================

static void CloseProcessHandles(PROCESS_INFORMATION& process)
{
    if (process.hProcess)
    {
        CloseHandle(process.hProcess);
        process.hProcess = nullptr;
    }

    if (process.hThread)
    {
        CloseHandle(process.hThread);
        process.hThread = nullptr;
    }
}


//====================================================
// Speak
//
// Generates a processed speech WAV file.
//
// Returns:
//
//     true  - Piper and FFmpeg succeeded.
//
//     false - Something failed or shutdown was
//             requested.
//====================================================

bool SpeechGenerator::GenerateSpeech
(
    const std::string& text,
    const std::atomic<bool>& stopRequested
)
{
    //================================================
    // PROJECT PATHS
    //================================================

    // Use the application's current working directory
    // as the project root.
    //
    // This replaces paths such as:
    //
    // Example of the old machine-specific path:
    // C:/dev/raylib-projects/TingoBingo_1.0/
    //
    // with paths relative to the project.
    const std::filesystem::path projectRoot =
        std::filesystem::current_path();

    const std::filesystem::path piperPath =
        projectRoot /
        "tools" /
        "piper" /
        "piper.exe";

    const std::filesystem::path piperModelPath =
        projectRoot /
        "tools" /
        "piper" /
        "en_GB-alan-medium.onnx";

    const std::filesystem::path ffmpegPath =
        projectRoot /
        "tools" /
        "ffmpeg" /
        "bin" /
        "ffmpeg.exe";

    const std::filesystem::path rawSpeechPath =
        projectRoot /
        "assets" /
        "audio" /
        "voice" /
        "speech_raw.wav";

    const std::filesystem::path speechPath =
        projectRoot /
        "assets" /
        "audio" /
        "voice" /
        "speech.wav";


    //================================================
    // VERIFY REQUIRED FILES
    //================================================

    // Check that Piper exists before trying to launch it.
    if (!std::filesystem::exists(piperPath))
    {
        std::cout
            << "Piper executable not found: "
            << piperPath.string()
            << "\n";

        return false;
    }


    // Check that the Piper voice model exists.
    if (!std::filesystem::exists(piperModelPath))
    {
        std::cout
            << "Piper model not found: "
            << piperModelPath.string()
            << "\n";

        return false;
    }


    // Check that FFmpeg exists.
    if (!std::filesystem::exists(ffmpegPath))
    {
        std::cout
            << "FFmpeg executable not found: "
            << ffmpegPath.string()
            << "\n";

        return false;
    }


    //================================================
    // PIPER INPUT PIPE
    //================================================

    // Create a Windows pipe that will become Piper's
    // standard input.
    //
    // Our program writes text into:
    //
    //     piperStdInWrite
    //
    // Piper reads that text from:
    //
    //     piperStdInRead
    SECURITY_ATTRIBUTES securityAttributes{};

    securityAttributes.nLength =
        sizeof(securityAttributes);

    securityAttributes.bInheritHandle = TRUE;

    securityAttributes.lpSecurityDescriptor =
        nullptr;


    HANDLE piperStdInRead = nullptr;
    HANDLE piperStdInWrite = nullptr;


    if (!CreatePipe(
            &piperStdInRead,
            &piperStdInWrite,
            &securityAttributes,
            0))
    {
        std::cout
            << "Failed to create Piper stdin pipe. Error: "
            << GetLastError()
            << "\n";

        return false;
    }


    // The parent process needs the write handle,
    // but Piper only needs the read handle.
    //
    // Prevent Piper from inheriting the write handle.
    //
    // This is important because otherwise Piper could
    // keep its own copy of the write side open.
    if (!SetHandleInformation(
            piperStdInWrite,
            HANDLE_FLAG_INHERIT,
            0))
    {
        std::cout
            << "Failed to configure Piper stdin pipe. Error: "
            << GetLastError()
            << "\n";

        CloseHandle(piperStdInRead);
        CloseHandle(piperStdInWrite);

        return false;
    }


    //================================================
    // START PIPER
    //================================================

    STARTUPINFOA piperStartupInfo{};
    PROCESS_INFORMATION piperProcess{};

    piperStartupInfo.cb =
        sizeof(piperStartupInfo);


    // Tell Windows that Piper should use the handles
    // supplied below for standard input/output/error.
    piperStartupInfo.dwFlags |=
        STARTF_USESTDHANDLES;


    // Piper reads its text from our pipe.
    piperStartupInfo.hStdInput =
        piperStdInRead;


    // Keep Piper's output/error behaviour normal.
    piperStartupInfo.hStdOutput =
        GetStdHandle(STD_OUTPUT_HANDLE);

    piperStartupInfo.hStdError =
        GetStdHandle(STD_ERROR_HANDLE);


    // Build Piper's command line.
    //
    // Notice that the user's text is NOT part of this
    // command line.
    //
    // That means characters such as:
    //
    //     "
    //     &
    //     |
    //     <
    //     >
    //
    // cannot be interpreted as shell commands.
    //
    // The text will be sent through stdin instead.
    std::string piperCommand =
        "\"" + piperPath.string() + "\" "
        "--model \"" + piperModelPath.string() + "\" "
        "--output_file \"" + rawSpeechPath.string() + "\"";


    // CreateProcessA() may modify the command-line
    // buffer, so commandLine must be a mutable string.
    if (!CreateProcessA(
            nullptr,
            piperCommand.data(),
            nullptr,
            nullptr,
            TRUE,
            CREATE_NO_WINDOW,
            nullptr,
            projectRoot.string().c_str(),
            &piperStartupInfo,
            &piperProcess))
    {
        std::cout
            << "Failed to launch Piper. Error: "
            << GetLastError()
            << "\n";

        CloseHandle(piperStdInRead);
        CloseHandle(piperStdInWrite);

        return false;
    }


    // Piper now owns the read side of the pipe.
    // The parent no longer needs it.
    CloseHandle(piperStdInRead);
    piperStdInRead = nullptr;


    //================================================
    // SEND TEXT TO PIPER
    //================================================

    // Add a newline so Piper receives the text as a
    // complete line, equivalent to the old:
    //
    // echo "text"
    //
    // approach.
    std::string input =
        text + "\n";


    DWORD bytesWritten = 0;


    if (!WriteFile(
            piperStdInWrite,
            input.c_str(),
            static_cast<DWORD>(input.size()),
            &bytesWritten,
            nullptr))
    {
        std::cout
            << "Failed to send text to Piper. Error: "
            << GetLastError()
            << "\n";

        CloseHandle(piperStdInWrite);
        piperStdInWrite = nullptr;

        TerminateProcess(
            piperProcess.hProcess,
            1
        );

        CloseProcessHandles(piperProcess);

        return false;
    }


    // Make sure the entire input was written.
    if (bytesWritten != input.size())
    {
        std::cout
            << "Piper stdin write was incomplete.\n";

        CloseHandle(piperStdInWrite);
        piperStdInWrite = nullptr;

        TerminateProcess(
            piperProcess.hProcess,
            1
        );

        CloseProcessHandles(piperProcess);

        return false;
    }


    // Closing the write side tells Piper:
    //
    // "There is no more input."
    //
    // Piper can now finish processing the text.
    CloseHandle(piperStdInWrite);
    piperStdInWrite = nullptr;


    //================================================
    // WAIT FOR PIPER
    //================================================

    // Don't use INFINITE here.
    //
    // Instead, check every 50 ms so the worker can
    // respond to a shutdown request.
    DWORD waitResult =
        WAIT_TIMEOUT;


    while (waitResult == WAIT_TIMEOUT)
    {
        if (stopRequested)
        {
            TerminateProcess(
                piperProcess.hProcess,
                1
            );

            CloseProcessHandles(piperProcess);

            return false;
        }


        waitResult =
            WaitForSingleObject(
                piperProcess.hProcess,
                50
            );
    }


    // If Windows reported a wait error, treat it as
    // a failed speech generation.
    if (waitResult != WAIT_OBJECT_0)
    {
        std::cout
            << "Failed while waiting for Piper. Error: "
            << GetLastError()
            << "\n";

        CloseProcessHandles(piperProcess);

        return false;
    }


    //================================================
    // CHECK PIPER RESULT
    //================================================

    DWORD piperExitCode = 0;

    if (!GetExitCodeProcess(
            piperProcess.hProcess,
            &piperExitCode))
    {
        std::cout
            << "Failed to get Piper exit code. Error: "
            << GetLastError()
            << "\n";

        CloseProcessHandles(piperProcess);

        return false;
    }


    if (piperExitCode != 0)
    {
        std::cout
            << "Piper failed. Exit code: "
            << piperExitCode
            << "\n";

        CloseProcessHandles(piperProcess);

        return false;
    }


    // Piper succeeded.
    CloseProcessHandles(piperProcess);


    //================================================
    // VERIFY PIPER OUTPUT
    //================================================

    // Make sure Piper actually created the WAV file
    // before asking FFmpeg to process it.
    if (!std::filesystem::exists(rawSpeechPath))
    {
        std::cout
            << "Piper completed but did not create: "
            << rawSpeechPath.string()
            << "\n";

        return false;
    }


    //================================================
    // START FFMPEG
    //================================================

    STARTUPINFOA ffmpegStartupInfo{};
    PROCESS_INFORMATION ffmpegProcess{};

    ffmpegStartupInfo.cb =
        sizeof(ffmpegStartupInfo);


    // Build the FFmpeg command line using paths relative
    // to the project location.
    //
    // No machine-specific C:\dev\... paths remain.
    std::string ffmpegCommand =
        "\"" + ffmpegPath.string() + "\" "
        "-y "
        "-i \"" + rawSpeechPath.string() + "\" "
        "-af \"aecho=0.8:0.88:60:0.25,"
        "highpass=f=250,"
        "lowpass=f=3500,"
        "acrusher=bits=12:mix=0.15\" "
        "\"" + speechPath.string() + "\"";


    if (!CreateProcessA(
            nullptr,
            ffmpegCommand.data(),
            nullptr,
            nullptr,
            FALSE,
            CREATE_NO_WINDOW,
            nullptr,
            projectRoot.string().c_str(),
            &ffmpegStartupInfo,
            &ffmpegProcess))
    {
        std::cout
            << "Failed to launch FFmpeg. Error: "
            << GetLastError()
            << "\n";

        return false;
    }


    //================================================
    // WAIT FOR FFMPEG
    //================================================

    waitResult =
        WAIT_TIMEOUT;


    while (waitResult == WAIT_TIMEOUT)
    {
        if (stopRequested)
        {
            TerminateProcess(
                ffmpegProcess.hProcess,
                1
            );

            CloseProcessHandles(ffmpegProcess);

            return false;
        }


        waitResult =
            WaitForSingleObject(
                ffmpegProcess.hProcess,
                50
            );
    }


    if (waitResult != WAIT_OBJECT_0)
    {
        std::cout
            << "Failed while waiting for FFmpeg. Error: "
            << GetLastError()
            << "\n";

        CloseProcessHandles(ffmpegProcess);

        return false;
    }


    //================================================
    // CHECK FFMPEG RESULT
    //================================================

    DWORD ffmpegExitCode = 0;

    if (!GetExitCodeProcess(
            ffmpegProcess.hProcess,
            &ffmpegExitCode))
    {
        std::cout
            << "Failed to get FFmpeg exit code. Error: "
            << GetLastError()
            << "\n";

        CloseProcessHandles(ffmpegProcess);

        return false;
    }


    if (ffmpegExitCode != 0)
    {
        std::cout
            << "FFmpeg failed. Exit code: "
            << ffmpegExitCode
            << "\n";

        CloseProcessHandles(ffmpegProcess);

        return false;
    }


    // FFmpeg succeeded.
    CloseProcessHandles(ffmpegProcess);


    //================================================
    // VERIFY FINAL OUTPUT
    //================================================

    // Make sure the final processed WAV actually exists
    // before telling SpeechController that it is ready.
    if (!std::filesystem::exists(speechPath))
    {
        std::cout
            << "FFmpeg completed but did not create: "
            << speechPath.string()
            << "\n";

        return false;
    }


    //================================================
    // SUCCESS
    //================================================


    return true;
}

    MyTransform3D localTransform{};

    localTransform.rotationAngle += localRotation;
    localTransform.scale = {1.0f, 1.0f, 1.0f};

    transform = MakeChildTransform(parentTransform, localTransform);
    
    hand.SetTransform(transform);
}

void ForeArm::AdvanceWave(float dt, ArmWaveState armWaveState)
{
    const float speed = 450.0f;
    const float minRotation = 0.0f;
    const float maxRotation = 60.0f;
    const float restingRotation = 0.0f;

    switch (armWaveState)
    {
    case ArmWaveState::Waving:
        switch (waveState)
        {
            case WaveState::None:
                waveState = WaveState::Raising;
                
                break;

            case WaveState::Raising:
                if(side == "left")
                {
                    localRotation += dt * speed;
                    
                    if(localRotation >= maxRotation)
                        waveState = WaveState::Lowering;
                }
                else if(side == "right")
                {
                    localRotation -= dt * speed;
                    
                    if(localRotation <= -maxRotation)
                        waveState = WaveState::Lowering;
                }
                break;

            case WaveState::Lowering:
                if(side == "left")
                {
                    localRotation -= dt * speed;
                    
                    if(localRotation <= minRotation)
                        waveState = WaveState::Raising;
                }
                else if(side == "right")
                {
                    localRotation += dt * speed;
                    
                    if(localRotation >= -minRotation)
                        waveState = WaveState::Raising;
                }

                break;

            default:
                break;
        }

        break;
        
    case ArmWaveState::None:
    {
        const float returnSpeed = 180.0f;

        if (localRotation > restingRotation)
        {
            localRotation -= dt * returnSpeed;

            if (localRotation < restingRotation)
                localRotation = restingRotation;
        }
        else if (localRotation < restingRotation)
        {
            localRotation += dt * returnSpeed;

            if (localRotation > restingRotation)
                localRotation = restingRotation;
        }

        if (localRotation == restingRotation)
            waveState = WaveState::None;

        break;
    }

    default:
        break;
    }

    SetTransform(tempParentTransform);
}

void UpperArm::AdvanceWave(float dt)
{
    if(armWaveState == ArmWaveState::None)
    {
        if 
        (
            state.waveState == TingoState::WaveState::WaveLeft && 
            side == "left"
        )
        {
            armWaveState = ArmWaveState::Raising;
        }
        else if 
        (
            state.waveState == TingoState::WaveState::WaveRight && 
            side == "right"
        )
        {
            armWaveState = ArmWaveState::Raising;
        }
    }

    const float speed = 300.0f;
    const float minRotation = 0.0f;
    const float maxRotation = 135.0f;

    switch (armWaveState)
    {
        case ArmWaveState::None:
            break;
        
        case ArmWaveState::Raising:
            if
            (
                state.waveState == TingoState::WaveState::WaveLeft && 
                side == "left"
            )
            {
                localRotation += dt * speed;
                if(localRotation >= maxRotation)
                {
                    localRotation = maxRotation;
                    armWaveState = ArmWaveState::Waving;
                }
            }

            if
            (
                state.waveState == TingoState::WaveState::WaveRight && 
                side == "right"
            )
            {
                localRotation -= dt * speed;
                if(localRotation <= -maxRotation)
                {
                    localRotation = -maxRotation;
                    armWaveState = ArmWaveState::Waving;
                }
            }
            break;

        case ArmWaveState::Waving:
            if(state.waveState == TingoState::WaveState::Inactive)
            {
                armWaveState = ArmWaveState::Lowering;
            }
            break;
            
        case ArmWaveState::Lowering:
            if(side == "left")
            {
                localRotation -= dt * speed;
                if(localRotation <= minRotation)
                {
                    localRotation = minRotation;
                    armWaveState = ArmWaveState::None;
                }
            }

            if(side == "right")
            {
                localRotation += dt * speed;
                if(localRotation >= minRotation)
                {
                    localRotation = minRotation;
                    armWaveState = ArmWaveState::None;
                }
            }

            break;

        default:
            break;
    }

    SetTransform(tempParentTransform);
}

ArmWaveState UpperArm::GetWaveState() const
{
    return armWaveState;
}

void Eye::AdvanceBlink(float dt)
{
    const float speed = 10.0f;
    const float minScale = 0.1f;
    const float maxScale = 1.0f;

    if (state.eyeStateBrain == TingoState::EyeStateBrain::Close)
    {
        if (eyeState != EyeState::Closed)
        {
            eyeState = EyeState::Closing;

            localScale.y -= dt * speed;

            if (localScale.y <= minScale)
            {
                localScale.y = minScale;
                eyeState = EyeState::Closed;
            }
        }
    }
    else if (state.eyeStateBrain == TingoState::EyeStateBrain::Open)
    {
        if (state.blinkState == TingoState::BlinkState::Blinking)
        {
            switch (eyeState)
            {
                case EyeState::Open:
                    eyeState = EyeState::Closing;
                    break;

                case EyeState::Closing:
                    localScale.y -= dt * speed;

                    if (localScale.y <= minScale)
                    {
                        localScale.y = minScale;
                        eyeState = EyeState::Opening;
                    }
                    break;

                case EyeState::Closed:
                    eyeState = EyeState::Opening;
                    break;

                case EyeState::Opening:
                    localScale.y += dt * speed;

                    if (localScale.y >= maxScale)
                    {
                        localScale.y = maxScale;
                        eyeState = EyeState::Open;
                    }
                    break;
            }
        }
        else
        {
            if (eyeState == EyeState::Closed)
            {
                eyeState = EyeState::Opening;
            }

            if (eyeState == EyeState::Opening ||
                eyeState == EyeState::Closing)
            {
                localScale.y += dt * speed;

                if (localScale.y >= maxScale)
                {
                    localScale.y = maxScale;
                    eyeState = EyeState::Open;
                }
            }
        }
    }

    SetTransform(tempParentTransform);
}

EyeState Eye::GetEyeState() const
{
    return eyeState;
}

#include "TingoBingo/Body/Head/MouthDisplay.h"
#include "raymath.h"
#include <iostream>

MouthDisplay::MouthDisplay(
    TingoAssets& assets, 
    TingoDimensions& dimensions,
    TingoState& state
) :
    assets(assets),
    dimensions(dimensions),
    state(state)
{}

void MouthDisplay::Initialise()
{
    float Yoffset = -dimensions.mouthDisplay.height / 2.0f - 
                    0.2f + dimensions.head.height / 2.0f;

    positionOffset = {
        0.0f,
        Yoffset,
        dimensions.head.length / 2.0f + dimensions.mouthDisplay.length / 2.0f        
    };

    mouthPlaneOffset = {0.0f, 0.0f, 0.06f};

    int frameWidth = 
        dimensions.mouthSpriteSheet.width / 
        dimensions.mouthSpriteSheet.cols;

    int frameHeight = 
        dimensions.mouthSpriteSheet.height /
        dimensions.mouthSpriteSheet.rows;
    
    int totalFrames = 
        dimensions.mouthSpriteSheet.rows *
        dimensions.mouthSpriteSheet.cols;
    
    float frameDuration = 0.4f;

        animation.Initialise
    (
        frameWidth, 
        frameHeight, 
        totalFrames, 
        dimensions.mouthSpriteSheet.cols, 
        frameDuration
    );

    // Generate a flat rectangular mesh for the mouth using 
    //the specified width and height.
    //The final two arguments (1, 1) specify one subdivision 
    //along the length and width.
    // LoadModelFromMesh() converts the generated mesh 
    //into a Model that Raylib can render.
    // The resulting model is stored in mouthPlane for
    //later positioning, texturing, and drawing.
    
    //- 0.05 is so plane is slightly smaller than display so weirdness
    //doesnt happen at edges
    mouthPlane = LoadModelFromMesh(GenMeshPlane(
        dimensions.mouthDisplay.width - 0.05,  // Width of the mouth plane
        dimensions.mouthDisplay.height - 0.05, // Height of the mouth plane
        1,                                     // Number of subdivisions along the width
        1                                      // Number of subdivisions along the length
    ));
    
    // Assign the mouth sprite sheet image as the diffuse texture
    // for the first material on the mouth plane.
    // The diffuse texture determines the image displayed on the surface
    // of the 3D model, allowing us to display Tingo's mouth sprites.
    mouthPlane.materials[0].maps[MATERIAL_MAP_DIFFUSE].texture =
        assets.mouthSpriteSheet;

    SetMouthFrame(0);
}

void MouthDisplay::Update(float dt)
{
    animation.Update(dt);

    SetMouthFrame(animation.GetFrame());

    AdvanceSpeakAnimation();
}

void MouthDisplay::Draw()
{
   // Rectangle source = animation.GetSourceRectangle();

    DrawModelEx(
        assets.mouthDisplay,
        transform.position,
        transform.rotationAxis,
        transform.rotationAngle,
        transform.scale,
        WHITE
    );

    DrawModelEx(
        mouthPlane,
        mouthPlaneTransform.position,
        mouthPlaneTransform.rotationAxis,
        mouthPlaneTransform.rotationAngle,
        mouthPlaneTransform.scale,
        WHITE
    );
}

void MouthDisplay::SetTransform(MyTransform3D parentTransform)
{
    MyTransform3D localTransform{};

    localTransform.position = positionOffset;
    localTransform.rotationAxis = {1.0f, 0.0f, 0.0f};
    localTransform.rotationAngle = 0.0f;
    localTransform.scale = {1.0f, 1.0f, 1.0f};

    transform = MakeChildTransform(parentTransform, localTransform);

    MyTransform3D localMouthPlaneTransform{};

    localMouthPlaneTransform.position = mouthPlaneOffset;
    localMouthPlaneTransform.rotationAxis = {1.0f, 0.0f, 0.0f};
    localMouthPlaneTransform.rotationAngle = 90.0f;
    localMouthPlaneTransform.scale = {1.0f, 1.0f, 1.0f};

    mouthPlaneTransform = 
        MakeChildTransform(transform, localMouthPlaneTransform);
}

void MouthDisplay::SetMouthFrame(int frame)
{
    int col = frame % dimensions.mouthSpriteSheet.cols;
    int row = frame / dimensions.mouthSpriteSheet.cols;

    float frameWidth =
    1.0f / dimensions.mouthSpriteSheet.cols;

    float frameHeight =
        1.0f / dimensions.mouthSpriteSheet.rows;

    float u0 = col * frameWidth;
    float u1 = u0 + frameWidth;

    float v0 = row * frameHeight;
    float v1 = v0 + frameHeight;

    float* texcoords = mouthPlane.meshes[0].texcoords;

    texcoords[0] = u0;
    texcoords[1] = v0;

    texcoords[2] = u1;
    texcoords[3] = v0;

    texcoords[4] = u0;
    texcoords[5] = v1;

    texcoords[6] = u1;
    texcoords[7] = v1;

    // Update the GPU's texture-coordinate buffer for the mouth mesh.
    // mouthPlane.meshes[0] selects the first mesh in the mouth model.
    // 1 identifies the texture-coordinate (UV) buffer.
    // texcoords contains the updated UV coordinates.
    // 8 * sizeof(float) specifies the number of bytes to upload
    // (8 floats, representing UV coordinates for 4 vertices).
    // 0 specifies the starting offset in the buffer.
    UpdateMeshBuffer(
        mouthPlane.meshes[0],
        1,
        texcoords,
        8 * sizeof(float),
        0
    );
}

void MouthDisplay::ShutDown()
{
    UnloadModel(mouthPlane);
}

void MouthDisplay::AdvanceSpeakAnimation()
{
    if (state.speaking)
    {
        if (!animation.IsPlaying())
        {
            animation.Play(1, 2, AnimationPriority::Idle);
        }
    }
    else
    {
        animation.Stop();

        if (state.brainState == TingoState::BrainState::Sleeping)
        {
            SetMouthFrame(1);
        }
        else
        {
            SetMouthFrame(0);
        }
    }
}

    transform = MakeChildTransform(parentTransform, localTransform);

    leftThigh.SetTransform(transform);
    rightThigh.SetTransform(transform);
}

        "assets/shaders/lighting.fs"
    );

    // Set the shader locations used by raylib.
    lightingShader.locs[SHADER_LOC_VECTOR_VIEW] =
        GetShaderLocation(lightingShader, "viewPos");

    // Set our custom uniform locations.
    lightDirectionLoc =
        GetShaderLocation(lightingShader, "lightDirection");

    ambientLoc =
        GetShaderLocation(lightingShader, "ambient");

    viewPositionLoc =
        GetShaderLocation(lightingShader, "viewPos");

    // Configure the directional light.
    //
    // The vector represents the direction in which light rays travel.
    // The shader reverses this direction when calculating illumination.
    float lightDirection[3] = {
        0.5f,
        -0.6f,
        -1.0f
    };

    SetShaderValue(
        lightingShader,
        lightDirectionLoc,
        lightDirection,
        SHADER_UNIFORM_VEC3
    );

    // Low-level ambient illumination keeps shadows readable.
    float ambient[3] = {
        0.40f,
        0.40f,
        0.40f
    };

    SetShaderValue(
        lightingShader,
        ambientLoc,
        ambient,
        SHADER_UNIFORM_VEC3
    );

        // Collect all the models that should use the lighting shader.
    Model* models[] = {
        &body,
        &pelvis,
        &thigh,
        &knee,
        &shin,
        &foot,
        &neck,

        &finger,
        &mouthDisplay,
        &eye,
        &pupil,
        &nose,
        &eyeBrow,
        &ear,
        &antenna
    };

    // Assign the shader to every material in every model.
    for (Model* model : models)
    {
        for (int i = 0; i < model->materialCount; i++)
        {
            model->materials[i].shader = lightingShader;
        }
    }
}

    tingo(tingo),
    state(state)

void TingoBrain::Initialise()
{
    state.brainState = TingoState::BrainState::Idle;
    idleTimer = 0.0f;

    state.waveState = TingoState::WaveState::Inactive;
    waveSequence = WaveSequence::Waiting;
    waveDurationTimer = 0.0f;
    idleWaveTimer = 0.0f;

    state.blinkState = TingoState::BlinkState::Inactive;
    blinkSequence = BlinkSequence::WaitingBetweenBlinks;
    blinkTimer = 0.0f;
    nextBlinkTimer = GetRandomValue(2, 6);

    sleepSequence = SleepSequence::Inactive;
    state.eyeStateBrain = TingoState::EyeStateBrain::Open;
    sleepTimer = 0.0f;
    snoreTimer = 0.0f;
    nextSnoreTimer = GetRandomValue(3, 6);

    state.speaking = false;

    randomPhraseTimer = GetRandomValue(10, 20);
}

void TingoBrain::Update(float dt)
{
    speechController.Update();

    randomPhraseTimer -= dt;

    if(randomPhraseTimer <= 0)
    {
        if(!state.speaking)
        {
            Speak(ChooseRandomPhrase());
            randomPhraseTimer = GetRandomValue(10, 20);
        }
    }

    if(speechController.SpeechFinished())
    {
        state.speaking = false;
    }
    else
    {
        state.speaking = true;
    }


    switch (state.brainState)
    {
        case TingoState::BrainState::Idle:
        {
            AdvanceIdleWave(dt);
            AdvanceBlink(dt);

            idleTimer += dt;

            if (idleTimer > 30.0f)
            {
                Speak("I am going to sleep");
                state.brainState = TingoState::BrainState::Sleeping;

                idleTimer = 0.0f;
                idleWaveTimer = 0.0f;
            }

            break;
        }

        case TingoState::BrainState::Sleeping:
        {
            AdvanceSleep(dt);
            break;
        }

        case TingoState::BrainState::Greeting:
        {
            AdvanceGreeting(dt);
            break;
        }

        default:
            break;
    }
}

}

void TingoBrain::AdvanceIdleWave(float dt)
{
    switch (waveSequence)
    {
        case WaveSequence::Waiting:
        {
            state.waveState = TingoState::WaveState::Inactive;
            idleWaveTimer += dt;

            if (idleWaveTimer >= 5.0f)
            {
                state.waveState = TingoState::WaveState::WaveRight;

                waveDurationTimer = 0.0f;
                idleWaveTimer = 0.0f;

                waveSequence = WaveSequence::RightArm;
            }

            break;
        }

        case WaveSequence::RightArm:
        {
            if (tingo.GetRightArmWaveState() == ArmWaveState::Waving)
            {
                waveDurationTimer += dt;

                if (waveDurationTimer >= 2.0f)
                {
                    state.waveState = TingoState::WaveState::Inactive;
                    waveSequence = WaveSequence::WaitingForRightArm;
                }
            }

            break;
        }

        case WaveSequence::WaitingForRightArm:
        {
            if (tingo.GetRightArmWaveState() == ArmWaveState::None)
            {
                idleWaveTimer = 0.0f;
                waveSequence = WaveSequence::WaitingBetweenArms;
            }

            break;
        }

        case WaveSequence::WaitingBetweenArms:
        {
            idleWaveTimer += dt;

            if (idleWaveTimer >= 2.0f)
            {
                state.waveState = TingoState::WaveState::WaveLeft;

                waveDurationTimer = 0.0f;
                idleWaveTimer = 0.0f;

                waveSequence = WaveSequence::LeftArm;
            }

            break;
        }

        case WaveSequence::LeftArm:
        {
            if (tingo.GetLeftArmWaveState() == ArmWaveState::Waving)
            {
                waveDurationTimer += dt;

                if (waveDurationTimer >= 2.0f)
                {
                    state.waveState = TingoState::WaveState::Inactive;
                    waveSequence = WaveSequence::WaitingForLeftArm;
                }
            }

            break;
        }

        case WaveSequence::WaitingForLeftArm:
        {
            if (tingo.GetLeftArmWaveState() == ArmWaveState::None)
            {
                idleWaveTimer = 0.0f;
                waveDurationTimer = 0.0f;

                waveSequence = WaveSequence::Waiting;
            }

            break;
        }

        default:
            break;
    }
}

void TingoBrain::AdvanceBlink(float dt)
{
    switch (blinkSequence)
    {
        case BlinkSequence::WaitingBetweenBlinks:
        {
            blinkTimer += dt;

            if (blinkTimer > nextBlinkTimer)
            {
                state.blinkState = TingoState::BlinkState::Blinking;
                blinkSequence = BlinkSequence::Blinking;

                blinkTimer = 0.0f;
                nextBlinkTimer = GetRandomValue(2, 6);
            }

            break;
        }

        case BlinkSequence::Blinking:
        {
            if (
                tingo.GetLeftEyeState() == EyeState::Open &&
                tingo.GetRightEyeState() == EyeState::Open
            )
            {
                state.blinkState = TingoState::BlinkState::Inactive;
                blinkSequence = BlinkSequence::WaitingBetweenBlinks;

                blinkTimer = 0.0f;
            }

            break;
        }

        default:
            break;
    }
}

void TingoBrain::AdvanceSleep(float dt)
{
    switch (sleepSequence)
    {
        case SleepSequence::Inactive:
        {
            sleepSequence = SleepSequence::PreparingToSleep;
            break;
        }

        case SleepSequence::PreparingToSleep:
        {
            state.eyeStateBrain = TingoState::EyeStateBrain::Open;

            sleepSequence = SleepSequence::WaitingForLimbs;
            break;
        }

        case SleepSequence::WaitingForLimbs:
        {
            AdvanceIdleWave(dt);
            AdvanceBlink(dt);

            if 
            (
                tingo.GetRightArmWaveState() == ArmWaveState::None &&
                tingo.GetLeftArmWaveState() == ArmWaveState::None
            )
            {
                state.waveState = TingoState::WaveState::Inactive;
                state.blinkState = TingoState::BlinkState::Inactive;
                state.eyeStateBrain = TingoState::EyeStateBrain::Close;
                sleepSequence = SleepSequence::ClosingEyes;
            }

            break;
        }

        case SleepSequence::ClosingEyes:
        {
            if (
                tingo.GetLeftEyeState() == EyeState::Closed &&
                tingo.GetRightEyeState() == EyeState::Closed
            )
            {
                sleepTimer = 0.0f;
                sleepSequence = SleepSequence::Sleeping;
            }

            break;
        }

        case SleepSequence::Sleeping:
        {
            sleepTimer += dt;
            snoreTimer += dt;

            if(snoreTimer >= nextSnoreTimer)
            {
                if(!state.speaking) 
                { 
                    Speak("Snore");
                }

                snoreTimer = 0.0f;
                nextSnoreTimer = GetRandomValue(3, 6);
            }

            if (sleepTimer > 30.0f)
            {
                sleepSequence = SleepSequence::WakingUp;
                sleepTimer = 0.0f;
            }

            break;
        }        

        case SleepSequence::WakingUp:
        {
            state.eyeStateBrain = TingoState::EyeStateBrain::Open;
            
            if 
            (
                tingo.GetLeftEyeState() == EyeState::Open &&
                tingo.GetRightEyeState() == EyeState::Open
            )
            {
                sleepSequence = SleepSequence::Finished;
            }
            
            break;
        }

        case SleepSequence::Finished:
        {
            // Reset the wave sequence.
            state.waveState = TingoState::WaveState::Inactive;
            waveSequence = WaveSequence::Waiting;
            waveDurationTimer = 0.0f;
            idleWaveTimer = 0.0f;

            // Reset the blink sequence.
            state.blinkState = TingoState::BlinkState::Inactive;
            blinkSequence = BlinkSequence::WaitingBetweenBlinks;
            blinkTimer = 0.0f;
            nextBlinkTimer = GetRandomValue(2, 6);

            // Return to idle.
            idleTimer = 0.0f;
            state.brainState = TingoState::BrainState::Idle;
            sleepSequence = SleepSequence::Inactive;

            Speak("I am waking up");

            break;
        }

    default:
        break;
    }
}

//====================================================
// ProjectDump.cpp
//
// Creates a text file containing the contents of all
// C++ source and header files in the project.
//
// This utility is useful for creating a single file
// that can be shared for code reviews or archived as
// a snapshot of the project.
//====================================================

#include <filesystem>
#include <fstream>
#include <iostream>
#include <string>

namespace fs = std::filesystem;

int main()
{
    // Create the output file.
    std::ofstream out("ProjectDump.txt");

    // Visit every file in the project directory.
    for (const auto& entry : fs::recursive_directory_iterator("."))
    {
        // Ignore directories.
        if (!entry.is_regular_file())
            continue;

        // Don't include the output file itself.
        if (entry.path().filename() == "ProjectDump.txt")
            continue;

        // Skip generated build files.
        if (entry.path().string().find("build") != std::string::npos)
            continue;

        std::string ext = entry.path().extension().string();

        // Only include C++ source and header files.
        if (ext == ".cpp" || ext == ".h")
        {
            // Write a heading for each file.
            out << "=====================================\n";
            out << entry.path().string() << "\n";
            out << "=====================================\n\n";

            std::ifstream in(entry.path());

            std::string line;

            // Copy the file into the output.
            while (std::getline(in, line))
            {
                out << line << '\n';
            }

            out << "\n\n";
        }
    }

    std::cout << "Finished!\n";

    return 0;
}

//====================================================
// ProjectTree.cpp
//
// Generates a text file showing the folder structure
// of the current project.
//
// This utility demonstrates how to use
// std::filesystem to recursively traverse directories
// and produce a simple tree view.
//====================================================

// iostream provides std::cout for writing to the console.
#include <iostream>

// fstream provides file input and output.
#include <fstream>

// filesystem provides classes and functions for working
// with files and directories.
#include <filesystem>

// Create a shorter alias so we can write "fs::"
// instead of "std::filesystem::".
namespace fs = std::filesystem;

//===============================================================
// PrintTree
//
// Recursively writes the contents of a directory to the
// output file as a tree structure.
//
// Parameters:
//   path  - The directory currently being scanned.
//   out   - The output file.
//   depth - The current directory depth, used to indent
//           child files and folders.
//===============================================================
void PrintTree(const fs::path& path, std::ofstream& out, int depth = 0)
{
    // Visit every file and folder in the current directory.
    for (const auto& entry : fs::directory_iterator(path))
    {
        // Skip the Git repository metadata.
        if (entry.is_directory() && entry.path().filename() == ".git")
            continue;

        // Indent according to the current folder depth.
        for (int i = 0; i < depth; i++)
            out << "│   ";

        // Write the file or directory name.
        out << "├── " << entry.path().filename().string();

        // Add a trailing '/' to make directories easy to identify.
        if (entry.is_directory())
            out << "/";

        out << '\n';

        // Recursively process child directories.
        if (entry.is_directory())
        {
            PrintTree(entry.path(), out, depth + 1);
        }
    }
}

int main()
{
    // "." represents the current working directory.
    fs::path root = ".";

    // Create the output file.
    std::ofstream out("FileStructure.txt");

    // Stop if the file couldn't be created.
    if (!out)
    {
        std::cout << "Couldn't create output file.\n";
        return 1;
    }

    // Write the root folder name.
    out << root.string() << '\n';

    // Generate the directory tree.
    PrintTree(root, out);

    // Inform the user that the operation completed.
    std::cout << "Done! Wrote FileStructure.txt\n";

    return 0;
}
