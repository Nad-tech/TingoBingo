#pragma once

#include "raylib.h"
#include "raymath.h"

struct MyTransform3D
{
    Vector3 position = { 0.0f, 0.0f, 0.0f };

    // Axis around which the model rotates.
    Vector3 rotationAxis = { 0.0f, 0.0f, 0.0f };

    // Rotation angle in degrees.
    float rotationAngle = 0.0f;

    // Scale along the X, Y and Z axes.
    Vector3 scale = { 1.0f, 1.0f, 1.0f };
};

inline MyTransform3D MakeChildTransform(
    const MyTransform3D& parent,
    const MyTransform3D& child)
{
    MyTransform3D result{};

    // 1. Scale the child's local position.
    Vector3 scaledOffset = Vector3Multiply(
        child.position,
        parent.scale
    );

    // 2. Convert the parent's rotation to a quaternion.
    Quaternion parentRotation = QuaternionFromAxisAngle(
        parent.rotationAxis,
        parent.rotationAngle * DEG2RAD
    );

    // 3. Rotate the child's offset around the parent.
    Vector3 rotatedOffset = Vector3RotateByQuaternion(
        scaledOffset,
        parentRotation
    );

    // 4. Calculate the child's world position.
    result.position = Vector3Add(
        parent.position,
        rotatedOffset
    );

    // 5. Combine rotations.
    Quaternion childRotation = QuaternionFromAxisAngle(
        child.rotationAxis,
        child.rotationAngle * DEG2RAD
    );

    Quaternion combinedRotation = QuaternionMultiply(
        parentRotation,
        childRotation
    );

    // 6. Convert the combined rotation back to axis-angle.
    Vector3 axis;
    float angle;

    QuaternionToAxisAngle(
        combinedRotation,
        &axis,
        &angle
    );

    result.rotationAxis = axis;
    result.rotationAngle = angle * RAD2DEG;

    // 7. Combine scales.
    result.scale = Vector3Multiply(
        parent.scale,
        child.scale
    );

    return result;
}

#include "Game.h"
#include "raylib.h"

void Game::Initialise()
{
    InitWindow(10, 10, "TingoBingo 3D");

    int monitor = GetCurrentMonitor();
    int width = GetMonitorWidth(monitor);
    int height = GetMonitorHeight(monitor);

    screenDimensions.screenWidth = width;
    screenDimensions.screenHeight = height;

    SetWindowSize(width, height);
    SetWindowPosition(0, 0);

    SetTargetFPS(60);
    DisableCursor();

    world.Initialise(screenDimensions);
}

void Game::Run()
{
    while (!WindowShouldClose())
    {
        float dt = GetFrameTime();

        world.Update(dt);

        BeginDrawing();

        ClearBackground(RAYWHITE);

        world.Draw();

        EndDrawing();
    }
}

void Game::Shutdown()
{
    world.Shutdown();

    CloseWindow();
}

#include "GameWorld.h"
#include <cmath>
#include "raymath.h"
#include "TingoBingo/TingoDimensions.h"

GameWorld::GameWorld() :
    tingo(assets, dimensions)
{}

inline bool dirFlag = true;

void GameWorld::Initialise(ScreenDimensions& screenDimensions)
{
    this->screenDimensions = screenDimensions;

    // Configure the 3D camera.
    camera.position = { -5.0, 10.0f, 20.0f };  // Move camera further away and higher // Camera's position in the 3D world (X, Y, Z)
    camera.target = { 0.0f, 0.0f, 0.0f };  // Point the camera looks at
    camera.up = { 0.0f, 1.0f, 0.0f };      // Defines the upward direction (positive Y)
    camera.fovy = 45.0f;                   // Vertical field of view in degrees
    camera.projection = CAMERA_PERSPECTIVE;// Objects farther away appear smaller

    assets.Initialise();
    dimensions.Initialise
    (
        assets.GetBodyDimensions(),
        assets.GetPelvisDimensions(),
        assets.GetThighDimensions(),
        assets.GetKneeDimensions(),
        assets.GetShinDimensions(),
        assets.GetFootDimensions(),
        assets.GetNeckDimensions(),
        assets.GetShoulderDimensions(),
        assets.GetHeadDimensions(),
        assets.GetUpperArmDimensions(),
        assets.GetForeArmDimensions(),
        assets.GetElbowDimensions(),
        assets.GetHandDimensions(),
        assets.GetFingerDimensions()
    );

    initialTransform.position = {
        0.0f, 
        dimensions.bodyShape.height / 2.0f +
        dimensions.pelvis.height +
        dimensions.thigh.height +
        dimensions.knee.height +
        dimensions.shin.height +
        dimensions.foot.height, 
        0.0f
    };

    initialTransform.rotationAxis = {0.0f, 1.0f, 0.0f};
    initialTransform.rotationAngle = 0.0f;
    initialTransform.scale = {1.0f, 1.0f, 1.0f};

    tingo.Initialise();
    tingo.SetTransform(initialTransform);
}

void GameWorld::Update(float dt)
{
    // Allow free movement around the 3D world.
    UpdateCamera(&camera, CAMERA_FREE);

    dirFlag ? initialTransform.rotationAngle += dt * 80 : initialTransform.rotationAngle -= dt * 80;
     

    if(initialTransform.rotationAngle <= 0.0) dirFlag = true;
    if(initialTransform.rotationAngle >= 720.0) dirFlag = false;    

    tingo.SetTransform(initialTransform);
    tingo.Update(dt);
}

// Draw a screen-space widget showing the camera's orientation.
void GameWorld::DrawOrientationGizmo()
{
    float x = screenDimensions.screenWidth;

    const Vector2 centre = {
    static_cast<float>(x - 100),
    100.0f
};
    const float axisLength = 35.0f;

    // Calculate the direction from the camera towards its target.
    Vector3 forward = Vector3Normalize(
        Vector3Subtract(camera.target, camera.position)
    );

    // Calculate the camera's right and up directions.
    Vector3 right = Vector3Normalize(
        Vector3CrossProduct(forward, camera.up)
    );

    Vector3 up = Vector3Normalize(
        Vector3CrossProduct(right, forward)
    );

    // World axes and their colours.
    const Vector3 axes[] = {
        { 1.0f, 0.0f, 0.0f },  // X
        { 0.0f, 1.0f, 0.0f },  // Y
        { 0.0f, 0.0f, 1.0f }   // Z
    };

    const Color colours[] = {
        RED,
        GREEN,
        BLUE
    };

    const char* labels[] = { "X", "Y", "Z" };

    // Draw each world axis from the widget's centre.
    for (int i = 0; i < 3; i++)
    {
        const Vector3& axis = axes[i];

        // Project the world axis onto the camera's screen directions.
        float screenX = Vector3DotProduct(axis, right);
        float screenY = -Vector3DotProduct(axis, up);

        Vector2 end = {
            centre.x + screenX * axisLength,
            centre.y + screenY * axisLength
        };

        DrawLineEx(centre, end, 3.0f, colours[i]);

        // Place the axis label just beyond the line's endpoint.
        Vector2 labelPosition = {
            end.x + screenX * 8.0f - 5.0f,
            end.y + screenY * 8.0f - 6.0f
        };

        DrawText(
            labels[i],
            static_cast<int>(labelPosition.x),
            static_cast<int>(labelPosition.y),
            14,
            colours[i]
        );
    }

    // Mark the widget's centre.
    DrawCircleV(centre, 3.0f, DARKGRAY);
}

void GameWorld::Draw()
{
    BeginMode3D(camera);

    // Room dimensions.
    const float roomWidth = 20.0f;
    const float roomHeight = 8.0f;
    const float roomDepth = 20.0f;

    // Half-dimensions make positioning the walls easier.
    const float halfWidth = roomWidth / 2.0f;
    const float halfDepth = roomDepth / 2.0f;

    // Floor.
    DrawPlane(
        { 0.0f, 0.0f, 0.0f },
        { roomWidth, roomDepth },
        LIGHTGRAY
    );

    // Front wall: positive Z.
    /*DrawCube(
        { 0.0f, roomHeight / 2.0f, halfDepth },
        roomWidth, roomHeight, 0.2f,
        GRAY
    );*/

    // Back wall wall: negative Z.
    DrawCube(
        { 0.0f, roomHeight / 2.0f, -halfDepth },
        roomWidth, roomHeight, 0.2f,
        GRAY
    );

    // Left wall: negative X.
    DrawCube(
        { -halfWidth, roomHeight / 2.0f, 0.0f },
        0.2f, roomHeight, roomDepth,
        DARKGRAY
    );

    // Right wall: positive X.
    DrawCube(
        { halfWidth, roomHeight / 2.0f, 0.0f },
        0.2f, roomHeight, roomDepth,
        DARKGRAY
    );

    DrawGrid(100, 1.0f);

    tingo.Draw();

    EndMode3D();

    DrawOrientationGizmo();
}

void Elbow::SetTransform(MyTransform3D parentTransform)
{
    MyTransform3D localTransform{};

    localTransform.position = positionOffset;
    localTransform.rotationAxis = {1.0f, 0.0f, 0.0f};
    localTransform.rotationAngle = 0.0f;

//====================================================
// ProjectDump.cpp
//
// Creates a text file containing the contents of all
// C++ source and header files in the project.
//
// This utility is useful for creating a single file
// that can be shared for code reviews or archived as
// a snapshot of the project.
//====================================================

#include <filesystem>
#include <fstream>
#include <iostream>
#include <string>

namespace fs = std::filesystem;

int main()
{
    // Create the output file.
    std::ofstream out("ProjectDump.txt");

    // Visit every file in the project directory.
    for (const auto& entry : fs::recursive_directory_iterator("."))
    {
        // Ignore directories.
        if (!entry.is_regular_file())
            continue;

        // Don't include the output file itself.
        if (entry.path().filename() == "ProjectDump.txt")
            continue;

        // Skip generated build files.
        if (entry.path().string().find("build") != std::string::npos)
            continue;

        std::string ext = entry.path().extension().string();

        // Only include C++ source and header files.
        if (ext == ".cpp" || ext == ".h")
        {
            // Write a heading for each file.
            out << "=====================================\n";
            out << entry.path().string() << "\n";
            out << "=====================================\n\n";

            std::ifstream in(entry.path());

            std::string line;

            // Copy the file into the output.
            while (std::getline(in, line))
            {
                out << line << '\n';
            }

            out << "\n\n";
        }
    }

    std::cout << "Finished!\n";

    return 0;
}
